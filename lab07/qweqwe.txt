// Struct to represent a 3D point
struct Point3D {
    float x, y, z;
    Point3D(float x = 0, float y = 0, float z = 0) : x(x), y(y), z(z) {}
    Point3D operator+(const Point3D& other) const { return Point3D(x + other.x, y + other.y, z + other.z); }
    Point3D operator*(float scalar) const { return Point3D(x * scalar, y * scalar, z * scalar); }
};

// Struct to represent a triangle
struct Triangle {
    Point3D v1, v2, v3;
    Triangle(const Point3D& v1, const Point3D& v2, const Point3D& v3) : v1(v1), v2(v2), v3(v3) {}
};

/**
 * MarchingCubes - A simple implementation of the marching cubes algorithm
 *
 * What is Marching Cubes?
 * - It's an algorithm to create 3D surfaces from 3D scalar fields (like medical scans)
 * - It works by examining each cube in a 3D grid and determining how the surface passes through it
 * - Each cube has 8 vertices, each with a scalar value
 * - The algorithm creates triangles where the surface intersects the cube edges
 */
class MarchingCubes {
private:
    int gridSize;                    // Size of the 3D grid (gridSize x gridSize x gridSize)
    float isoValue;                  // The value that defines the surface
    std::vector<std::vector<std::vector<float>>> grid;  // 3D grid of scalar values
    std::vector<Triangle> triangles; // Generated triangles

    // Cube vertices positions (relative to cube origin)
    static const Point3D cubeVertices[8];

    // Edge connections - which vertices each edge connects
    static const int edgeConnections[12][2];

public:
    /**
     * Constructor
     * @param size - Size of the 3D grid
     * @param iso - Iso value (surface threshold)
     */
    MarchingCubes(int size, float iso) : gridSize(size), isoValue(iso) {
        // Initialize the 3D grid
        grid.resize(gridSize);
        for (int i = 0; i < gridSize; i++) {
            grid[i].resize(gridSize);
            for (int j = 0; j < gridSize; j++) {
                grid[i][j].resize(gridSize);
            }
        }

        // Fill the grid with a simple sphere function for demonstration
        generateSphereField();
    }

    /**
     * Generate a simple sphere scalar field for testing
     * This creates a sphere in the center of the grid
     */
    void generateSphereField() {
        float center = gridSize / 2.0f;
        float radius = gridSize / 4.0f;

        for (int x = 0; x < gridSize; x++) {
            for (int y = 0; y < gridSize; y++) {
                for (int z = 0; z < gridSize; z++) {
                    // Calculate distance from center
                    float dx = x - center;
                    float dy = y - center;
                    float dz = z - center;
                    float distance = sqrt(dx*dx + dy*dy + dz*dz);

                    // Store the distance as the scalar value
                    grid[x][y][z] = radius - distance;
                }
            }
        }
    }

    /**
     * Get the scalar value at a grid position
     */
    float getGridValue(int x, int y, int z) const {
        if (x < 0 || x >= gridSize || y < 0 || y >= gridSize || z < 0 || z >= gridSize) {
            return -1.0f; // Outside grid is considered "outside" the surface
        }
        return grid[x][y][z];
    }

    /**
     * Interpolate between two points based on their scalar values
     * This finds where exactly the surface crosses an edge
     */
    Point3D interpolateVertex(const Point3D& p1, const Point3D& p2, float val1, float val2) const {
        if (abs(isoValue - val1) < 0.00001f) return p1;
        if (abs(isoValue - val2) < 0.00001f) return p2;
        if (abs(val1 - val2) < 0.00001f) return p1;

        // Linear interpolation
        float t = (isoValue - val1) / (val2 - val1);
        return p1 + (p2 + p1 * -1.0f) * t;
    }

    /**
     * Process a single cube and generate triangles
     * This is the core of the marching cubes algorithm
     */
    void processCube(int x, int y, int z) {
        // Get the 8 corner values of the cube
        float cubeValues[8];
        Point3D cubePoints[8];

        for (int i = 0; i < 8; i++) {
            Point3D vertex = cubeVertices[i];
            cubePoints[i] = Point3D(x + vertex.x, y + vertex.y, z + vertex.z);
            cubeValues[i] = getGridValue(x + vertex.x, y + vertex.y, z + vertex.z);
        }

        // Determine which vertices are inside the surface
        int cubeIndex = 0;
        for (int i = 0; i < 8; i++) {
            if (cubeValues[i] > isoValue) {
                cubeIndex |= (1 << i);
            }
        }

        // If all vertices are inside or outside, no surface passes through
        if (edgeTable[cubeIndex] == 0) return;

        // Find intersection points on edges
        Point3D edgePoints[12];
        if (edgeTable[cubeIndex] & 1) {
            edgePoints[0] = interpolateVertex(cubePoints[0], cubePoints[1], cubeValues[0], cubeValues[1]);
        }
        if (edgeTable[cubeIndex] & 2) {
            edgePoints[1] = interpolateVertex(cubePoints[1], cubePoints[2], cubeValues[1], cubeValues[2]);
        }
        if (edgeTable[cubeIndex] & 4) {
            edgePoints[2] = interpolateVertex(cubePoints[2], cubePoints[3], cubeValues[2], cubeValues[3]);
        }
        if (edgeTable[cubeIndex] & 8) {
            edgePoints[3] = interpolateVertex(cubePoints[3], cubePoints[0], cubeValues[3], cubeValues[0]);
        }
        if (edgeTable[cubeIndex] & 16) {
            edgePoints[4] = interpolateVertex(cubePoints[4], cubePoints[5], cubeValues[4], cubeValues[5]);
        }
        if (edgeTable[cubeIndex] & 32) {
            edgePoints[5] = interpolateVertex(cubePoints[5], cubePoints[6], cubeValues[5], cubeValues[6]);
        }
        if (edgeTable[cubeIndex] & 64) {
            edgePoints[6] = interpolateVertex(cubePoints[6], cubePoints[7], cubeValues[6], cubeValues[7]);
        }
        if (edgeTable[cubeIndex] & 128) {
            edgePoints[7] = interpolateVertex(cubePoints[7], cubePoints[4], cubeValues[7], cubeValues[4]);
        }
        if (edgeTable[cubeIndex] & 256) {
            edgePoints[8] = interpolateVertex(cubePoints[0], cubePoints[4], cubeValues[0], cubeValues[4]);
        }
        if (edgeTable[cubeIndex] & 512) {
            edgePoints[9] = interpolateVertex(cubePoints[1], cubePoints[5], cubeValues[1], cubeValues[5]);
        }
        if (edgeTable[cubeIndex] & 1024) {
            edgePoints[10] = interpolateVertex(cubePoints[2], cubePoints[6], cubeValues[2], cubeValues[6]);
        }
        if (edgeTable[cubeIndex] & 2048) {
            edgePoints[11] = interpolateVertex(cubePoints[3], cubePoints[7], cubeValues[3], cubeValues[7]);
        }

        // Create triangles using the lookup table
        // Note: For simplicity, we're using a basic implementation
        // In practice, you'd use the complete triTable
        for (int i = 0; triTable[cubeIndex][i] != -1; i += 3) {
            if (i + 2 < 16 && triTable[cubeIndex][i + 2] != -1) {
                Triangle tri(
                    edgePoints[triTable[cubeIndex][i]],
                    edgePoints[triTable[cubeIndex][i + 1]],
                    edgePoints[triTable[cubeIndex][i + 2]]
                );
                triangles.push_back(tri);
            }
        }
    }

    /**
     * Generate the mesh by processing all cubes
     */
    void generateMesh() {
        triangles.clear();

        // Process each cube in the grid
        for (int x = 0; x < gridSize - 1; x++) {
            for (int y = 0; y < gridSize - 1; y++) {
                for (int z = 0; z < gridSize - 1; z++) {
                    processCube(x, y, z);
                }
            }
        }

        std::cout << "Generated " << triangles.size() << " triangles" << std::endl;
    }

    /**
     * Get the generated triangles
     */
    const std::vector<Triangle>& getTriangles() const {
        return triangles;
    }
};

};

    {0, 4}, {1, 5}, {2, 6}, {3, 7}   // Vertical edges
};

// OpenGL rendering class
class Renderer {
private:
    GLuint VAO, VBO;
    GLuint shaderProgram;
    std::vector<float> vertices;

    // Simple vertex shader source
    const char* vertexShaderSource = R"(
        #version 330 core
        layout (location = 0) in vec3 aPos;

        uniform mat4 model;
        uniform mat4 view;
        uniform mat4 projection;

        void main() {
            gl_Position = projection * view * model * vec4(aPos, 1.0);
        }
    )";

    // Simple fragment shader source
    const char* fragmentShaderSource = R"(
        #version 330 core
        out vec4 FragColor;

        void main() {
            FragColor = vec4(0.8, 0.3, 0.3, 1.0);  // Red color
        }
    )";

    GLuint compileShader(const char* source, GLenum type) {
        GLuint shader = glCreateShader(type);
        glShaderSource(shader, 1, &source, nullptr);
        glCompileShader(shader);

        // Check for compilation errors
        int success;
        char infoLog[512];
        glGetShaderiv(shader, GL_COMPILE_STATUS, &success);
        if (!success) {
            glGetShaderInfoLog(shader, 512, nullptr, infoLog);
            std::cerr << "Shader compilation failed: " << infoLog << std::endl;
        }

        return shader;
    }

public:
    void initialize() {
        // Create and compile shaders
        GLuint vertexShader = compileShader(vertexShaderSource, GL_VERTEX_SHADER);
        GLuint fragmentShader = compileShader(fragmentShaderSource, GL_FRAGMENT_SHADER);

        // Create shader program
        shaderProgram = glCreateProgram();
        glAttachShader(shaderProgram, vertexShader);
        glAttachShader(shaderProgram, fragmentShader);
        glLinkProgram(shaderProgram);

        // Clean up shaders
        glDeleteShader(vertexShader);
        glDeleteShader(fragmentShader);

        // Generate VAO and VBO
        glGenVertexArrays(1, &VAO);
        glGenBuffers(1, &VBO);

        // Setup vertex attributes
        glBindVertexArray(VAO);
        glBindBuffer(GL_ARRAY_BUFFER, VBO);
        glVertexAttribPointer(0, 3, GL_FLOAT, GL_FALSE, 3 * sizeof(float), (void*)0);
        glEnableVertexAttribArray(0);

        glBindBuffer(GL_ARRAY_BUFFER, 0);
        glBindVertexArray(0);
    }

    void updateMesh(const std::vector<Triangle>& triangles) {
        vertices.clear();

        // Convert triangles to vertex array
        for (const auto& tri : triangles) {
            vertices.push_back(tri.v1.x);
            vertices.push_back(tri.v1.y);
            vertices.push_back(tri.v1.z);

            vertices.push_back(tri.v2.x);
            vertices.push_back(tri.v2.y);
            vertices.push_back(tri.v2.z);

            vertices.push_back(tri.v3.x);
            vertices.push_back(tri.v3.y);
            vertices.push_back(tri.v3.z);
        }

        // Update VBO
        glBindBuffer(GL_ARRAY_BUFFER, VBO);
        glBufferData(GL_ARRAY_BUFFER, vertices.size() * sizeof(float), vertices.data(), GL_STATIC_DRAW);
        glBindBuffer(GL_ARRAY_BUFFER, 0);
    }

    void render(float time) {
        // Clear the screen
        glClear(GL_COLOR_BUFFER_BIT | GL_DEPTH_BUFFER_BIT);

        // Use shader program
        glUseProgram(shaderProgram);

        // Set up matrices
        glm::mat4 model = glm::mat4(1.0f);
        model = glm::rotate(model, time * 0.5f, glm::vec3(0.0f, 1.0f, 0.0f));
        model = glm::rotate(model, time * 0.3f, glm::vec3(1.0f, 0.0f, 0.0f));
        model = glm::translate(model, glm::vec3(-15.0f, -15.0f, -15.0f)); // Center the mesh

        glm::mat4 view = glm::lookAt(
            glm::vec3(0.0f, 0.0f, 50.0f),  // Camera position
            glm::vec3(0.0f, 0.0f, 0.0f),   // Look at origin
            glm::vec3(0.0f, 1.0f, 0.0f)    // Up vector
        );

        glm::mat4 projection = glm::perspective(glm::radians(45.0f), 800.0f / 600.0f, 0.1f, 100.0f);

        // Set uniforms
        glUniformMatrix4fv(glGetUniformLocation(shaderProgram, "model"), 1, GL_FALSE, glm::value_ptr(model));
        glUniformMatrix4fv(glGetUniformLocation(shaderProgram, "view"), 1, GL_FALSE, glm::value_ptr(view));
        glUniformMatrix4fv(glGetUniformLocation(shaderProgram, "projection"), 1, GL_FALSE, glm::value_ptr(projection));

        // Draw triangles
        glBindVertexArray(VAO);
        glDrawArrays(GL_TRIANGLES, 0, vertices.size() / 3);
        glBindVertexArray(0);
    }

    void cleanup() {
        glDeleteVertexArrays(1, &VAO);
        glDeleteBuffers(1, &VBO);
        glDeleteProgram(shaderProgram);
    }
};

// Main function
int main() {
    // Initialize GLFW
    if (!glfwInit()) {
        std::cerr << "Failed to initialize GLFW" << std::endl;
        return -1;
    }

    // Set OpenGL version
    glfwWindowHint(GLFW_CONTEXT_VERSION_MAJOR, 3);
    glfwWindowHint(GLFW_CONTEXT_VERSION_MINOR, 3);
    glfwWindowHint(GLFW_OPENGL_PROFILE, GLFW_OPENGL_CORE_PROFILE);

    // Create window
    GLFWwindow* window = glfwCreateWindow(800, 600, "Marching Cubes Demo", nullptr, nullptr);
    if (!window) {
        std::cerr << "Failed to create GLFW window" << std::endl;
        glfwTerminate();
        return -1;
    }

    glfwMakeContextCurrent(window);

    // Initialize GLEW
    if (glewInit() != GLEW_OK) {
        std::cerr << "Failed to initialize GLEW" << std::endl;
        return -1;
    }

    // Enable depth testing
    glEnable(GL_DEPTH_TEST);

    // Set clear color
    glClearColor(0.1f, 0.1f, 0.1f, 1.0f);

    // Create marching cubes instance
    std::cout << "Creating marching cubes with 30x30x30 grid..." << std::endl;
    MarchingCubes marchingCubes(30, 0.0f);  // 30x30x30 grid, iso value 0

    // Generate mesh
    std::cout << "Generating mesh..." << std::endl;
    marchingCubes.generateMesh();

    // Initialize renderer
    Renderer renderer;
    renderer.initialize();
    renderer.updateMesh(marchingCubes.getTriangles());

    std::cout << "Rendering... Press ESC to exit" << std::endl;

    // Main render loop
    while (!glfwWindowShouldClose(window)) {
        // Handle input
        if (glfwGetKey(window, GLFW_KEY_ESCAPE) == GLFW_PRESS) {
            glfwSetWindowShouldClose(window, true);
        }

        // Render
        float time = glfwGetTime();
        renderer.render(time);

        // Swap buffers and poll events
        glfwSwapBuffers(window);
        glfwPollEvents();
    }

    // Cleanup
    renderer.cleanup();
    glfwTerminate();

    return 0;
}
